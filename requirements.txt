# ============================================================================
# 安装说明和注意事项
# ============================================================================
#
# 1. 环境要求:
#    - Python 3.10
#    - CUDA 12.4 (与PyTorch 2.6.0+cu124匹配)
#    - 推荐使用 conda 环境管理
#
# 2. 安装顺序 (重要):
#    conda create -n grpo3d python=3.10
#    conda activate grpo3d
#    
#    # 先安装PyTorch
#    pip install torch==2.6.0 torchvision==0.21.0 --index-url https://download.pytorch.org/whl/cu124
#    
#    # 再安装3D相关依赖 (顺序重要)
#    pip install kaolin==0.18.0 -f https://nvidia-kaolin.s3.us-east-2.amazonaws.com/torch-2.4.0_cu121.html
#    pip install spconv-cu124==2.3.8
#    
#    # 最后安装其他包
#    pip install -r requirements.txt
#
# 3. 环境变量配置 (关键):
#    export ATTN_BACKEND=xformers     # 强制使用xformers (避免flash-attn问题)
#    export HF_HUB_OFFLINE=1         # 使用本地模型
#    export CUDA_VISIBLE_DEVICES=0   # 指定GPU
#
# 4. 使用建议:
#    # 推荐使用项目提供的脚本
#    ./scripts/run_trellis.sh python your_script.py
#
# 5. 已知问题:
#    - flash-attn 2.8.2 与 PyTorch 2.6.0 有C++符号链接冲突
#    - 临时解决方案: 使用 xformers 替代 flash-attn
#    - kaolin 安装链接可能需要根据PyTorch版本调整
#
# 6. 验证安装:
#    python -c "import torch, xformers, spconv, kaolin; print('✅ 核心依赖安装成功')"
#    ./scripts/run_trellis.sh python scripts/test_trellis_with_model.py
